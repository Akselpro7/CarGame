function updateDriving(dt) { let accel = 0; if (keys["w"] || keys["arrowup"]) accel = 28; if (keys["s"] || keys["arrowdown"]) accel = -18; if (keys[" "]) velocity *= 0.92; if ((keys["shift"] || keys["shiftleft"]) && nitro > 0 && accel > 0) { accel *= 1.8; nitro = Math.max(0, nitro - dt * 25); } else { nitro = Math.min(100, nitro + dt * 8); } velocity += (accel - velocity * 1.8) * dt; velocity = Math.max(-25, Math.min(55, velocity)); let steerInput = 0; if (keys["a"] || keys["arrowleft"]) steerInput = 1; if (keys["d"] || keys["arrowright"]) steerInput = -1; steer += (steerInput - steer) * dt * 6; yaw += steer * velocity * dt * 0.04; // Corrected movement direction car.position.x += Math.sin(yaw) * velocity * dt; car.position.z += Math.cos(yaw) * velocity * dt; car.rotation.y = yaw; car.position.y = 0; }
